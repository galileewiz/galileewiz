Total Sales & Orders by Category
orders.sql · query_3.sql
SELECT
    category,
    COUNT(order_id) AS total_orders,
    SUM(quantity * price) AS total_revenue,
    ROUND(AVG(price), 2) AS avg_price
FROM
    orders
GROUP BY
    category
ORDER BY
    total_revenue DESC,
    category ASC;

Top Spending Customers
orders.sql · query_5.sql
SELECT
    customer_name,
    COUNT(order_id) AS total_orders,
    SUM(price) AS total_spent
FROM
    orders
GROUP BY
    customer_name
HAVING
    SUM(price) > 50
ORDER BY
    total_spent DESC,
    customer_name ASC;
