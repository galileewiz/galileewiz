expenses = [
    ("Office Rent",      25000),
    ("Electricity Bill", 3200),
    ("Team Outing",      5200),
    ("Software Sub",      999),
    ("Flight",           12000),
    ("Laptop Repair",    3500),
]

running_total = 0
print(f"{'#':<4} {'Description':<22} {'Amount':>10}  {'Running Total':>14}")
print("-" * 58)

for i, (desc, amount) in enumerate(expenses, start=1):
    running_total += amount
    print(f"{i:<4} {desc:<22} ₹{amount:>8,}  ₹{running_total:>12,}")

def check_budget(category, spent, budget):
    """Check if spending is within budget and print the status."""
    remaining    = budget - spent
    percent_used = (spent / budget) * 100

    if spent > budget:
        status = "❌ OVERSPENT"
    elif percent_used >= 80:
        status = "⚠️  WARNING "
    else:
        status = "✅ ON TRACK "

    print(f"{status}  {category:<18}"
          f"  Spent ₹{spent:,} / ₹{budget:,}"
          f"  ({percent_used:.0f}%)  Remaining ₹{remaining:,}")

check_budget("Rent",           25000, 25000)
check_budget("Food",            9250,  8000)
check_budget("Travel",         17300, 20000)
check_budget("Software",        4497,  5000)
check_budget("Office Supplies",  970,   2000)

import pandas as pd

df = pd.read_csv('expenses.csv')
print(f"Dataset: {len(df)} rows, {len(df.columns)} columns\n")

# ── Step 1: Total spend by category ──────────────
print("Total Spend by Category:")
category_total = (df.groupby('category')['amount']
                    .sum()
                    .sort_values(ascending=False))
print(category_total)

# ── Step 2: Filter only Paid expenses ────────────
paid_df = df[df['status'] == 'Paid']
print(f"\nPaid expenses : {len(paid_df)} rows")
print(f"Pending       : {len(df) - len(paid_df)} rows")

# ── Step 3: Top 5 highest single expenses ────────
print("\nTop 5 Highest Expenses:")
top5 = df.nlargest(5, 'amount')[['description', 'category', 'amount']]
print(top5.to_string(index=False))
